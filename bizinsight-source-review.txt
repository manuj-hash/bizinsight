const getDashboardAnalytics = async (req, res) => {
    try {
        const [
            ordersResult,
            productsResult,
            customersResult
        ] = await Promise.all([
            supabase
                .from("orders")
                .select("id, total_amount, status, ordered_at"),

            supabase
                .from("products")
                .select("id, stock_quantity, reorder_level, is_active"),

            supabase
                .from("customers")
                .select("id", { count: "exact", head: true })
        ]);

        if (ordersResult.error) throw ordersResult.error;
        if (productsResult.error) throw productsResult.error;
        if (customersResult.error) throw customersResult.error;

        const orders = ordersResult.data || [];
        const products = productsResult.data || [];

        const completedOrders = orders.filter(
            order => order.status === "completed"
        );

        const totalSales = completedOrders.reduce(
            (sum, order) => sum + Number(order.total_amount || 0),
            0
        );

        const lowStockProducts = products.filter(
            product =>
                product.is_active !== false &&
                Number(product.stock_quantity) <= Number(product.reorder_level)
        );

        const now = new Date();
        const currentMonthStart = new Date(
            now.getFullYear(),
            now.getMonth(),
            1
        );

        const previousMonthStart = new Date(
            now.getFullYear(),
            now.getMonth() - 1,
            1
        );

        const currentMonthSales = completedOrders
            .filter(order => new Date(order.ordered_at) >= currentMonthStart)
            .reduce((sum, order) => sum + Number(order.total_amount || 0), 0);

        const previousMonthSales = completedOrders
            .filter(order => {
                const date = new Date(order.ordered_at);
                return date >= previousMonthStart && date < currentMonthStart;
            })
            .reduce((sum, order) => sum + Number(order.total_amount || 0), 0);

        const growthPercent = previousMonthSales === 0
            ? null
            : Number(
                (
                    ((currentMonthSales - previousMonthSales) /
                        previousMonthSales) *
                    100
                ).toFixed(2)
            );

        res.status(200).json({
            totalProducts: products.length,
            totalCustomers: customersResult.count || 0,
            totalOrders: orders.length,
            completedOrders: completedOrders.length,
            totalSales: Number(totalSales.toFixed(2)),
            lowStockCount: lowStockProducts.length,
            lowStockProducts,
            currentMonthSales: Number(currentMonthSales.toFixed(2)),
            previousMonthSales: Number(previousMonthSales.toFixed(2)),
            growthPercent
        });
    } catch (error) {
        res.status(500).json({
            message: error.message
        });
    }
};

                    date.getMonth() + 1
                ).padStart(2, "0")}`;

                monthlySales[month] =
                    (monthlySales[month] || 0) +
                    Number(order.total_amount || 0);
            });

        res.status(200).json({
            monthlySales: Object.entries(monthlySales).map(
                ([month, sales]) => ({
                    month,
                    sales: Number(sales.toFixed(2))
                })
            ),
            orders: data || []
        });
    } catch (error) {
        res.status(500).json({
            message: error.message
        });
    }
};

            profile: profile,
            token: data.session.access_token,

const createOrder = async (req, res) => {
    try {
        const { customer_id = null, items, tax_amount = 0, discount_amount = 0, status = "pending" } = req.body;
        if (!Array.isArray(items) || items.length === 0) return res.status(400).json({ message: "At least one order item is required" });
        const seen = new Set();
        for (const item of items) {
            if (!item.product_id || !Number.isInteger(Number(item.quantity)) || Number(item.quantity) <= 0)
                return res.status(400).json({ message: "Each item requires product_id and a positive integer quantity" });
            if (seen.has(item.product_id)) return res.status(400).json({ message: "Duplicate products are not allowed in one order" });
            seen.add(item.product_id);
        }
        const tax = Number(tax_amount), discount = Number(discount_amount);
        if (!Number.isFinite(tax) || !Number.isFinite(discount) || tax < 0 || discount < 0)
            return res.status(400).json({ message: "Invalid tax or discount amount" });
        if (!["pending", "processing", "completed", "cancelled"].includes(status))
            return res.status(400).json({ message: "Invalid order status" });

        const orderNumber = `ORD-${Date.now()}-${require("crypto").randomBytes(3).toString("hex").toUpperCase()}`;
        const { data: orderId, error: rpcError } = await supabase.rpc("create_bizinsight_order", {

            p_tax_amount: tax,
            p_discount_amount: discount,
            p_status: status
        });
        if (rpcError) throw rpcError;

        const { data: completeOrder, error: fetchError } = await supabase
            .from("orders").select("*, customers(name, email), order_items(id, product_id, quantity, unit_price, line_total, products(name, sku))")

// Validate the Supabase access token AND load the trusted role from profiles.
// Never authorize using user_metadata: users can edit their own user metadata.
const protect = async (req, res, next) => {
    try {
        const header = req.headers.authorization || "";
        if (!header.startsWith("Bearer ")) {
            return res.status(401).json({ message: "Authorization token missing" });
        }
        const token = header.slice(7).trim();
        if (!token) return res.status(401).json({ message: "Authorization token missing" });

        const { data: authData, error: authError } = await supabase.auth.getUser(token);
        if (authError || !authData.user) {
            return res.status(401).json({ message: "Invalid or expired token" });
        }

        const { data: profile, error: profileError } = await supabase
            .from("profiles")
            .select("id, full_name, phone, role")
            .eq("id", authData.user.id)
            .maybeSingle();

        if (profileError) throw profileError;
        if (!profile || !["admin", "staff"].includes(profile.role)) {
            return res.status(403).json({ message: "An active BizInsight staff account is required" });
        }

        req.user = authData.user;
        req.profile = profile;
        next();
    } catch (error) {
        console.error("Authentication error:", error.message);
        return res.status(401).json({ message: "Authentication failed" });
    }
};

const allowRoles = (...roles) => (req, res, next) => {
    if (!req.profile || !roles.includes(req.profile.role)) {
        return res.status(403).json({ message: "You do not have permission to perform this action" });
    }
    next();
};

-- Creates order + items and decrements stock in one database transaction.
-- Execute permission is limited to service_role; the backend calls this RPC using
-- the server-side Supabase secret key after checking the user's staff/admin role.
CREATE OR REPLACE FUNCTION public.create_bizinsight_order(
  p_order_number text,
  p_customer_id uuid,
  p_items jsonb,
  p_tax_amount numeric DEFAULT 0,
  p_discount_amount numeric DEFAULT 0,
  p_status text DEFAULT 'pending'
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE
  v_order_id uuid;
  v_subtotal numeric := 0;
  v_tax numeric := COALESCE(p_tax_amount, 0);
  v_discount numeric := COALESCE(p_discount_amount, 0);
  v_item jsonb;
  v_product public.products%ROWTYPE;
  v_quantity integer;
BEGIN
  IF jsonb_typeof(p_items) <> 'array' OR jsonb_array_length(p_items) = 0 THEN
    RAISE EXCEPTION 'At least one order item is required';
  END IF;
  IF v_tax < 0 OR v_discount < 0 THEN RAISE EXCEPTION 'Tax and discount cannot be negative'; END IF;
  IF p_status NOT IN ('pending', 'processing', 'completed', 'cancelled') THEN RAISE EXCEPTION 'Invalid order status'; END IF;

  -- Validate and lock products in stable ID order to reduce concurrent deadlocks.
  FOR v_item IN SELECT value FROM jsonb_array_elements(p_items) ORDER BY (value->>'product_id')
  LOOP
    v_quantity := (v_item->>'quantity')::integer;
    IF v_quantity <= 0 THEN RAISE EXCEPTION 'Quantity must be a positive integer'; END IF;
    SELECT * INTO v_product FROM public.products
      WHERE id = (v_item->>'product_id')::uuid FOR UPDATE;
    IF NOT FOUND THEN RAISE EXCEPTION 'Product not found'; END IF;
    IF NOT v_product.is_active THEN RAISE EXCEPTION 'Product is inactive'; END IF;
    IF v_product.stock_quantity < v_quantity THEN RAISE EXCEPTION 'Insufficient stock for product %', v_product.name; END IF;
    v_subtotal := v_subtotal + (v_product.price * v_quantity);
  END LOOP;

  IF v_discount > v_subtotal + v_tax THEN RAISE EXCEPTION 'Discount cannot exceed subtotal plus tax'; END IF;

  INSERT INTO public.orders(order_number, customer_id, status, subtotal, tax_amount, discount_amount, total_amount)
  VALUES (p_order_number, p_customer_id, p_status, v_subtotal, v_tax, v_discount, v_subtotal + v_tax - v_discount)
  RETURNING id INTO v_order_id;

  FOR v_item IN SELECT value FROM jsonb_array_elements(p_items)
  LOOP
    v_quantity := (v_item->>'quantity')::integer;
    SELECT * INTO v_product FROM public.products WHERE id = (v_item->>'product_id')::uuid FOR UPDATE;
    UPDATE public.products SET stock_quantity = stock_quantity - v_quantity WHERE id = v_product.id;
    INSERT INTO public.order_items(order_id, product_id, quantity, unit_price)
    VALUES (v_order_id, v_product.id, v_quantity, v_product.price);
  END LOOP;

  RETURN v_order_id;
END;
$$;
REVOKE ALL ON FUNCTION public.create_bizinsight_order(text, uuid, jsonb, numeric, numeric, text) FROM PUBLIC, anon, authenticated;
GRANT EXECUTE ON FUNCTION public.create_bizinsight_order(text, uuid, jsonb, numeric, numeric, text) TO service_role;

// Category report
function displayCategoryReport() {
    const categoryData = {};

    products.forEach(product => {
        const category = product.category || "Uncategorized";
        const stock = Number(product.stock_quantity || 0);

        if (!categoryData[category]) {
            categoryData[category] = {
                count: 0,
                stock: 0
            };
        }

        categoryData[category].count++;
        categoryData[category].stock += stock;
    });

    const table = document.getElementById("categoryTable");
    const categories = Object.entries(categoryData);

    if (categories.length === 0) {
        table.innerHTML = `
            <tr>
                <td colspan="4">No product data available.</td>

        return;
    }

    table.innerHTML = categories.map(([category, data]) => {
        let status = "Good";

        if (data.stock === 0) {
            status = "Out of Stock";
        } else if (data.stock <= 20) {
            status = "Low Stock";

        return `

                <td>${data.count}</td>
                <td>${data.stock}</td>

}

// Escape HTML
function escapeHTML(value) {
    return String(value ?? "").replace(/[&<>"']/g, char => ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#039;"
    })[char]);
}

// CSV download
function downloadCSV(filename, rows) {
    if (!rows || rows.length === 0) {
        showMessage("No data available to download.", true);
        return;
    }

    const csv = rows.map(row =>
        row.map(value =>
            `"${String(value ?? "").replace(/"/g, '""')}"`
        ).join(",")
    ).join("\n");

    const blob = new Blob(
        ["\uFEFF" + csv],
        { type: "text/csv;charset=utf-8;" }
    );

    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");

    link.href = url;
    link.download = filename;
    link.click();

    URL.revokeObjectURL(url);
    showMessage(`${filename} downloaded successfully.`);
}

        ]);

            product.sku || "",
            product.category || "",

        ]);
    });

    downloadCSV("BizInsight_Inventory_Report.csv", rows);
}
